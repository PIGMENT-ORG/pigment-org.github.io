Browser Examples (Vanilla JS)

1. Get API Key

async function getApiKey(email) {
    const response = await fetch('https://pigment-api.onrender.com/v1/users', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email })
    });
    
    if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
    }
    
    const data = await response.json();
    return data.api_key;
}

// Usage
getApiKey('user@example.com')
    .then(key => console.log('API Key:', key))
    .catch(err => console.error('Error:', err));

2. Create Work

async function createWork(apiKey, title, genome, tags = []) {
    const response = await fetch('https://pigment-api.onrender.com/v1/works', {
        method: 'POST',
        headers: {
            'X-API-Key': apiKey,
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({
            title,
            content: genome,
            tags
        })
    });
    
    return await response.json();
}

4. Complete Workflow Example

async function runCompleteWorkflow() {
    try {
        // 1. Get API key
        const apiKey = await getApiKey('demo@example.com');
        console.log('API Key obtained');
        
        // 2. Create initial work
        const work = await createWork(apiKey, 'JS Demo', genome, ['javascript']);
        console.log('Work created:', work.id);
        
        // 3. Evolve it
        const evolved = await evolveWork(apiKey, work.id, 500, 1.2);
        console.log(`Evolved: ${evolved.fitness_before}% → ${evolved.fitness_after}%`);
        console.log('Operator used:', evolved.operator);
        
        // 4. Get kinship
        const kinshipResponse = await fetch('https://pigment-api.onrender.com/v1/kinship', {
            method: 'POST',
            headers: {
                'X-API-Key': apiKey,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({ work_id: evolved.child_id })
        });
        const kinship = await kinshipResponse.json();
        console.log('Parents:', kinship.parents?.length || 0);
        
        // 5. Apply prompt
        const promptResponse = await fetch('https://pigment-api.onrender.com/v1/prompt', {
            method: 'POST',
            headers: {
                'X-API-Key': apiKey,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({
                work_id: evolved.child_id,
                prompt: 'make it glitchy with neon colors'
            })
        });
        const promptResult = await promptResponse.json();
        console.log('AI operations:', promptResult.operations_applied);
        
        return promptResult.child_id;
        
    } catch (error) {
        console.error('Workflow failed:', error);
    }
}

Node.js with Fetch

// Node.js 18+ has native fetch
const API_URL = 'https://pigment-api.onrender.com';

class PigmentClient {
    constructor(apiKey) {
        this.apiKey = apiKey;
        this.baseUrl = API_URL;
    }
    
    async request(endpoint, options = {}) {
        const response = await fetch(`${this.baseUrl}${endpoint}`, {
            ...options,
            headers: {
                'X-API-Key': this.apiKey,
                'Content-Type': 'application/json',
                ...options.headers
            }
        });
        
        if (!response.ok) {
            const error = await response.json().catch(() => ({}));
            throw new Error(error.message || `HTTP ${response.status}`);
        }
        
        return response.json();
    }
    
    async createWork(title, content, tags = []) {
        return this.request('/v1/works', {
            method: 'POST',
            body: JSON.stringify({ title, content, tags })
        });
    }
    
    async evolve(workId, steps = 500, mutationRate = 1.0) {
        return this.request('/v1/evolve', {
            method: 'POST',
            body: JSON.stringify({ work_id: workId, steps, mutation_rate: mutationRate })
        });
    }
    
    async getKinship(workId) {
        return this.request('/v1/kinship', {
            method: 'POST',
            body: JSON.stringify({ work_id: workId })
        });
    }
}

// Usage
const client = new PigmentClient('your-api-key');
const work = await client.createWork('Node.js Example', genome);
const evolved = await client.evolve(work.id);

WebSocket Real-time Updates

// Connect to WebSocket for real-time evolution updates
const userId = 'user_' + Date.now().toString(36);
const ws = new WebSocket(`wss://pigment-api.onrender.com/ws/${userId}`);

ws.onopen = () => {
    console.log('Connected to PIGMENT WebSocket');
    
    // Authenticate
    ws.send(JSON.stringify({
        type: 'auth',
        api_key: 'your-api-key'
    }));
};

ws.onmessage = (event) => {
    const data = JSON.parse(event.data);
    
    switch(data.type) {
        case 'evolution_tick':
            console.log(`Progress: ${Math.round(data.progress * 100)}%`);
            console.log(`Generation ${data.generation}: ${data.fitness}% fitness`);
            console.log(`Operator: ${data.operator}`);
            break;
            
        case 'evolution_done':
            console.log('Evolution complete!');
            console.log(`Final fitness: ${data.fitness_after}%`);
            console.log(`Improvement: +${data.improvement}%`);
            ws.close();
            break;
            
        case 'fitness_milestone':
            console.log(`🏆 Milestone: ${data.milestone}% fitness!`);
            break;
            
        case 'error':
            console.error('WebSocket error:', data.error);
            break;
    }
};

ws.onclose = () => {
    console.log('Disconnected from WebSocket');
};

// Subscribe to a job
ws.send(JSON.stringify({
    type: 'subscribe_job',
    job_id: 'job_abc123'
}));

Error Handling with Retries

async function fetchWithRetry(url, options, maxRetries = 3) {
    for (let i = 0; i < maxRetries; i++) {
        try {
            const response = await fetch(url, options);
            
            if (response.status === 429) {
                // Rate limited - exponential backoff
                const waitTime = Math.pow(2, i) * 1000 + Math.random() * 1000;
                console.log(`Rate limited. Retrying in ${waitTime/1000}s...`);
                await new Promise(r => setTimeout(r, waitTime));
                continue;
            }
            
            if (!response.ok) {
                const error = await response.json().catch(() => ({}));
                throw new Error(error.message || `HTTP ${response.status}`);
            }
            
            return await response.json();
            
        } catch (err) {
            if (i === maxRetries - 1) throw err;
            if (err.name === 'TypeError') {
                // Network error, retry
                await new Promise(r => setTimeout(r, 1000 * Math.pow(2, i)));
                continue;
            }
            throw err;
        }
    }
}
