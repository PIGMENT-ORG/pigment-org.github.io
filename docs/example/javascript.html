JavaScript Examples

Complete JavaScript/Node.js examples for all PIGMENT API endpoints.

Installation

npm install pigment-api

Basic Setup

Initialize client 
import { PigmentClient } from 'pigment-api';

const client = new PigmentClient({
    apiKey: process.env.PIGMENT_API_KEY,
    environment: 'production' // or 'development'
});

Evolve an Image

// Evolve an image from URL
async function evolveFromUrl() {
    try {
        const result = await client.evolve({
            target_url: 'https://example.com/image.jpg',
            generations: 1000,
            polygons: 100,
            mutation_rate: 0.1,
            callback_url: 'https://your-server.com/webhook'
        });
        
        console.log('Evolution started:', result.job_id);
        console.log('Estimated time:', result.estimated_seconds, 'seconds');
        
        // Poll for results
        const status = await client.getJobStatus(result.job_id);
        console.log('Status:', status);
        
    } catch (error) {
        console.error('Error:', error.message);
    }
}

// Evolve from local file
async function evolveFromFile() {
    const fs = require('fs');
    const fileBuffer = fs.readFileSync('./my-image.jpg');
    
    const result = await client.evolve({
        target_image: fileBuffer.toString('base64'),
        generations: 5000
    });
}

Kinship Analysis

// Compare artworks
async function compareArtworks() {
    const result = await client.kinship({
        works: [
            'work_id_1',
            'work_id_2',
            'work_id_3'
        ],
        include_features: true,
        min_similarity: 0.5
    });
    
    console.log('Similarity matrix:', result.matrix);
    console.log('Family tree:', result.tree);
    
    // Visualize with included tool
    const viz = new KinshipViz('#container', result.tree);
}

// Find ancestors
async function findAncestors(workId) {
    const family = await client.kinship.lineage({
        work_id: workId,
        generations: 5
    });
    
    console.log('Ancestors:', family.ancestors);
    console.log('Descendants:', family.descendants);
}

Gallery Management

// List gallery
async function listGallery() {
    const works = await client.gallery.list({
        limit: 50,
        offset: 0,
        sort: 'fitness:desc',
        tags: ['abstract', 'portrait']
    });
    
    works.forEach(work => {
        console.log(`${work.id}: ${work.fitness}% - ${work.created_at}`);
    });
}

// Upload to gallery
async function uploadToGallery() {
    const result = await client.gallery.upload({
        image: imageBase64,
        title: 'My Masterpiece',
        tags: ['digital', 'evolutionary'],
        public: true
    });
    
    console.log('Gallery URL:', result.url);
}

// Batch operations
async function batchUpload() {
    const files = ['image1.jpg', 'image2.jpg', 'image3.jpg'];
    
    const results = await Promise.all(
        files.map(file => client.gallery.upload({
            image: fs.readFileSync(file).toString('base64'),
            title: file
        }))
    );
    
    console.log(`Uploaded ${results.length} files`);
}

Real-time Updates

// WebSocket connection
import { WebSocket } from 'ws';

const ws = new WebSocket('wss://api.pigment.dev/v1/ws', {
    headers: {
        'Authorization': `Bearer ${apiKey}`
    }
});

ws.on('open', () => {
    console.log('Connected to PIGMENT');
    
    // Subscribe to job updates
    ws.send(JSON.stringify({
        type: 'subscribe',
        job_id: 'job_123456'
    }));
});

ws.on('message', (data) => {
    const update = JSON.parse(data);
    
    switch(update.type) {
        case 'progress':
            console.log(`Progress: ${update.percent}%`);
            console.log(`Current fitness: ${update.fitness}`);
            break;
            
        case 'complete':
            console.log('Evolution complete!');
            console.log('Result:', update.result);
            ws.close();
            break;
            
        case 'error':
            console.error('Error:', update.message);
            break;
    }
});

ws.on('close', () => {
    console.log('Disconnected');
});

React Component Example

import React, { useState, useEffect } from 'react';
import { usePigment } from 'pigment-api/react';

function EvolutionViewer({ targetImage }) {
    const { evolve, job, progress } = usePigment();
    const [currentImage, setCurrentImage] = useState(null);
    
    useEffect(() => {
        if (targetImage) {
            evolve({
                target_image: targetImage,
                generations: 1000
            });
        }
    }, [targetImage]);
    
    useEffect(() => {
        if (job?.status === 'complete') {
            setCurrentImage(job.result.image);
        }
    }, [job]);
    
    return (
        

            

                

                    
Target

                    
                

                

                    
Evolved

                    {currentImage ? (
                        
                    ) : (
                        

                            

                            {progress}%
                        

                    )}
                

            

            

                
Generation: {job?.generation || 0}

                
Fitness: {job?.fitness?.toFixed(2)}%

                
Best: {job?.best_fitness?.toFixed(2)}%

            

        

    );
}

Error Handling

import { PigmentError } from 'pigment-api';

async function safeEvolve() {
    try {
        const result = await client.evolve(params);
        return result;
        
    } catch (error) {
        if (error instanceof PigmentError) {
            switch(error.code) {
                case 'AUTH_ERROR':
                    console.error('Invalid API key');
                    break;
                case 'RATE_LIMIT':
                    console.error('Rate limited. Retry in', error.retryAfter, 'seconds');
                    await sleep(error.retryAfter * 1000);
                    return safeEvolve();
                case 'INVALID_PARAMS':
                    console.error('Invalid parameters:', error.details);
                    break;
                default:
                    console.error('API error:', error.message);
            }
        } else {
            console.error('Network error:', error);
        }
    }
}
