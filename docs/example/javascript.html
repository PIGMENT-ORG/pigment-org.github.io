JavaScript Examples

Complete JavaScript/Node.js examples for all PIGMENT API endpoints.

Installation

npm install pigment-api

Basic Setup

import { PigmentClient } from 'pigment-api';

const client = new PigmentClient({
    apiKey: process.env.PIGMENT_API_KEY
});

Evolve Image

async function evolveFromUrl() {
    const job = await client.evolve({
        target_url: 'https://example.com/image.jpg',
        generations: 1000,
        polygons: 100
    });

    // Poll for results
    let status;
    do {
        await new Promise(r => setTimeout(r, 1000));
        status = await client.getJobStatus(job.job_id);
        console.log(`Progress: ${status.progress}%`);
    } while (status.status === 'processing');

    console.log('Complete!', status.result.image_url);
}

Using Prompts

async function generateWithPrompt() {
    const result = await client.prompt({
        prompt: 'a cyberpunk cityscape with neon lights',
        style: 'geometric',
        iterations: 500
    });
}

Kinship Analysis

async function compareWorks() {
    const result = await client.kinship.compare({
        work_ids: ['work_123', 'work_456'],
        include_features: true
    });
}
