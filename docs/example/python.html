import requests
import time
import json

# Configuration
API_URL = "https://pigment-api.onrender.com"
API_KEY = None  # Will be set after registration

def get_api_key(email):
    """Register a new user and get API key"""
    response = requests.post(
        f"{API_URL}/v1/users",
        json={"email": email}
    )
    response.raise_for_status()
    data = response.json()
    return data["api_key"]

def create_work(api_key, title, content, tags=None):
    """Create a new artwork"""
    response = requests.post(
        f"{API_URL}/v1/works",
        headers={"X-API-Key": api_key},
        json={
            "title": title,
            "content": content,
            "tags": tags or []
        }
    )
    response.raise_for_status()
    return response.json()

def evolve_work(api_key, work_id, steps=500, mutation_rate=1.0):
    """Evolve an artwork"""
    response = requests.post(
        f"{API_URL}/v1/evolve",
        headers={"X-API-Key": api_key},
        json={
            "work_id": work_id,
            "steps": steps,
            "mutation_rate": mutation_rate
        }
    )
    response.raise_for_status()
    return response.json()

def get_kinship(api_key, work_id):
    """Get kinship relationships"""
    response = requests.post(
        f"{API_URL}/v1/kinship",
        headers={"X-API-Key": api_key},
        json={"work_id": work_id}
    )
    response.raise_for_status()
    return response.json()

def search_similar(api_key, work_id, limit=5):
    """Find similar works"""
    response = requests.post(
        f"{API_URL}/v1/search/similar",
        headers={"X-API-Key": api_key},
        json={"work_id": work_id, "limit": limit}
    )
    response.raise_for_status()
    return response.json()

def apply_prompt(api_key, work_id, prompt):
    """Apply AI prompt to artwork"""
    response = requests.post(
        f"{API_URL}/v1/prompt",
        headers={"X-API-Key": api_key},
        json={"work_id": work_id, "prompt": prompt}
    )
    response.raise_for_status()
    return response.json()

def list_gallery(api_key, limit=10):
    """Browse public gallery"""
    response = requests.get(
        f"{API_URL}/v1/gallery",
        headers={"X-API-Key": api_key},
        params={"limit": limit}
    )
    response.raise_for_status()
    return response.json()

def main():
    # 1. Get API key
    print("Registering user...")
    api_key = get_api_key("user@example.com")
    print(f"API Key: {api_key}")
    
    # 2. Sample genome
    genome = '''-- PIGMENT Genome v6.0.0
-- @fitness 0.0

canvas { width: 400 height: 400 }

palette { color_0: #ff3366 }

layer base {
  zone poly_0 {
    color: palette.color_0
    opacity: 0.8
    points: 120,140 180,120 160,200
  }
}'''
    
    # 3. Create work
    print("\nCreating artwork...")
    work = create_work(api_key, "Python Example", genome, ["python", "demo"])
    print(f"Created work: {work['id']}")
    
    # 4. Evolve it
    print("\nEvolving artwork...")
    evolved = evolve_work(api_key, work['id'], steps=500)
    print(f"Evolved: {evolved['fitness_before']}% → {evolved['fitness_after']}%")
    print(f"Best operator: {evolved['operator']}")
    
    # 5. Check kinship
    print("\nChecking kinship...")
    kinship = get_kinship(api_key, evolved['child_id'])
    print(f"Parents: {len(kinship.get('parents', []))}")
    print(f"Children: {len(kinship.get('children', []))}")
    print(f"Siblings: {len(kinship.get('siblings', []))}")
    
    # 6. Apply prompt
    print("\nApplying AI prompt...")
    prompt_result = apply_prompt(api_key, evolved['child_id'], "make it glitchy")
    print(f"AI intent: {prompt_result.get('intent', 'N/A')}")
    print(f"Operations: {prompt_result.get('operations_applied', [])}")
    
    # 7. Browse gallery
    print("\nBrowsing gallery...")
    gallery = list_gallery(api_key, limit=5)
    print(f"Total works: {gallery['total']}")
    for work in gallery['works'][:3]:
        print(f"  - {work['title']}: {work['fitness']}% ({work['style']})")

if __name__ == "__main__":
    main()

Async Python with aiohttp

import aiohttp
import asyncio

async def evolve_async():
    async with aiohttp.ClientSession() as session:
        async with session.post(
            "https://pigment-api.onrender.com/v1/evolve",
            headers={"X-API-Key": "your-key"},
            json={"work_id": "work_abc123", "steps": 500}
        ) as response:
            return await response.json()

# Run
result = asyncio.run(evolve_async())

Error Handling Example

import requests
from requests.exceptions import RequestException
import time

def safe_api_call(func, *args, max_retries=3, **kwargs):
    """Make API call with retry logic"""
    for attempt in range(max_retries):
        try:
            return func(*args, **kwargs)
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 429:  # Rate limited
                wait = (2 ** attempt) + 1
                print(f"Rate limited. Waiting {wait}s...")
                time.sleep(wait)
                continue
            elif e.response.status_code == 401:
                print("Authentication error - check your API key")
                break
            else:
                print(f"HTTP Error: {e}")
                break
        except RequestException as e:
            print(f"Network error: {e}")
            break
    return None
