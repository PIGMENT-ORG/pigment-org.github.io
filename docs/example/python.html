Python Examples

Complete Python examples for all PIGMENT API endpoints.

Installation

pip install pigment-api

Basic Setup

import os
from pigment import PigmentClient

client = PigmentClient(
    api_key=os.environ.get('PIGMENT_API_KEY')
)

Evolve Image

import time

def evolve_image():
    job = client.evolve(
        target_url='https://example.com/image.jpg',
        generations=1000
    )

    while True:
        status = client.get_job_status(job['job_id'])
        print(f"Progress: {status['progress']}%")
        
        if status['status'] != 'processing':
            break
        time.sleep(1)

    print(f"Result: {status['result']['image_url']}")

Using Prompts

def generate_with_prompt():
    result = client.prompt(
        prompt='a serene landscape with mountains',
        style='realistic',
        iterations=500
    )
    return result

Kinship Analysis

def analyze_family_tree():
    lineage = client.kinship.get_lineage(
        work_id='work_123',
        generations=3
    )
    
    for ancestor in lineage['ancestors']:
        print(f"Ancestor: {ancestor['id']} - "
              f"Similarity: {ancestor['similarity']}")

Async Support

import asyncio
from pigment import AsyncPigmentClient

async def async_example():
    client = AsyncPigmentClient(api_key=os.environ.get('PIGMENT_API_KEY'))
    result = await client.evolve(target_url='https://example.com/image.jpg')
    return result
